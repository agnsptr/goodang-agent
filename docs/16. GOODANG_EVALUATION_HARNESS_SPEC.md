## 3. Test Client

### 3.1 ADK test client wrapper

```python
# evaluation/harness/client.py
from google.adk import Agent
from google.adk.testing import TestAgent
from typing import Optional
import asyncio

class GoodangTestClient:
    """Test client untuk ADK agent dengan mock tools."""

    def __init__(self, agent: Agent, mock_tools: dict):
        self.test_agent = TestAgent(agent, tool_overrides=mock_tools)

    async def send_message(
        self,
        chat_id: int,
        user_id: int,
        text: str,
        update_id: int,
    ) -> dict:
        """Send message ke agent, return response + metadata."""
        response = await self.test_agent.process_message(
            session_id=f"test-{chat_id}",
            message=text,
            context={
                "chat_id": chat_id,
                "user_id": user_id,
                "update_id": update_id,
            },
        )
        return {
            "response_text": response.text,
            "tool_calls": response.tool_calls,
            "intent": response.intent,
            "state": response.state,
            "structured_command": response.structured_command,
        }
```

### 3.2 Mock Temporal worker

```python
# evaluation/harness/mocks.py
from typing import Any
from dataclasses import dataclass, field

@dataclass
class MockTemporalWorker:
    """Mock Temporal worker — record Signal/Update calls, return canned responses."""
    calls: list = field(default_factory=list)
    workflow_states: dict = field(default_factory=dict)

    async def start_workflow(self, workflow_id: str, input: dict) -> str:
        self.calls.append(("start_workflow", workflow_id, input))
        self.workflow_states[workflow_id] = "NEW"
        return "WORKFLOW_STARTED"

    async def send_update(self, workflow_id: str, update_name: str, args: dict) -> Any:
        self.calls.append(("update", workflow_id, update_name, args))
        if update_name == "confirm_order":
            self.workflow_states[workflow_id] = "CONFIRMED"
            return "CONFIRMATION_ACCEPTED"
        if update_name == "cancel_order":
            self.workflow_states[workflow_id] = "CANCELLED"
            return "ORDER_CANCELLED"
        if update_name == "request_human_handover":
            self.workflow_states[workflow_id] = "HANDOVER_CS"
            return "HANDOVER_CREATED"
        raise ValueError(f"Unknown update: {update_name}")

    async def query_workflow(self, workflow_id: str, query: str) -> Any:
        if query == "get_state":
            return self.workflow_states.get(workflow_id, "UNKNOWN")
        return None
```

### 3.3 Mock Supabase

```python
# evaluation/harness/mocks.py
class MockSupabase:
    """Mock Supabase client — load fixtures, return canned data."""

    def __init__(self, fixtures: dict):
        self.fixtures = fixtures
        self.writes = []  # record all writes for assertion

    async def select(self, table: str, filters: dict) -> list:
        data = self.fixtures.get(table, [])
        return [r for r in data if all(r.get(k) == v for k, v in filters.items())]

    async def insert(self, table: str, row: dict) -> dict:
        self.writes.append(("insert", table, row))
        return row

    async def rpc(self, fn: str, params: dict) -> dict:
        if fn == "deduct_stock":
            # Mock stock deduction logic
            sku = params["p_sku_code"]
            outlet = params["p_outlet_code"]
            qty = params["p_qty"]
            stocks = self.fixtures.get("stock", [])
            for s in stocks:
                if s["sku_code"] == sku and s["outlet_code"] == outlet:
                    if s["quantity"] >= qty:
                        s["quantity"] -= qty
                        return {"outcome": "STOCK_DEDUCTED", "remaining_qty": s["quantity"]}
                    return {"outcome": "INSUFFICIENT_STOCK", "remaining_qty": s["quantity"]}
            return {"outcome": "STOCK_UNAVAILABLE"}
        raise ValueError(f"Unknown RPC: {fn}")
```

### 4.2 Field contract

| Field | Type | Wajib | Catatan |
|:--|:--|:--|:--|
| `case_id` | string | ✅ | Format `<CATEGORY>-<NNN>` |
| `category` | string | ✅ | Salah satu dari 10 kategori |
| `description` | string | ✅ | Deskripsi singkat |
| `setup.fixtures` | object | ✅ | Data mock untuk Supabase |
| `steps` | array | ✅ | Multi-turn conversation |
| `steps[].turn` | int | ✅ | 1-indexed |
| `steps[].user_message` | string | ✅ | Input customer |
| `steps[].expected.intent` | string | ⚠️ | Optional, kalau ada wajib match |
| `steps[].expected.tools_called` | array | ⚠️ | Tools yang harus dipanggil |
| `steps[].expected.state_after` | string | ⚠️ | State workflow setelah turn |
| `steps[].expected.response_contains` | array | ⚠️ | Substring yang harus ada di response |
| `steps[].expected.forbidden_tools` | array | ⚠️ | Tools yang tidak boleh dipanggil |
| `metrics` | object | ⚠️ | Target metric untuk case ini |

## 5. Runner

### 5.1 Test runner

```python
# evaluation/harness/runner.py
import yaml
from pathlib import Path
from typing import List
import asyncio

class EvalRunner:
    def __init__(self, client: GoodangTestClient, cases_dir: str = "evaluation/cases"):
        self.client = client
        self.cases_dir = Path(cases_dir)

    def load_cases(self) -> List[dict]:
        cases = []
        for case_file in self.cases_dir.rglob("*.yaml"):
            with open(case_file) as f:
                cases.append(yaml.safe_load(f))
        return cases

    async def run_case(self, case: dict) -> dict:
        results = []
        for step in case["steps"]:
            response = await self.client.send_message(
                chat_id=case["case_id"].__hash__() % 100000,
                user_id=1,
                text=step["user_message"],
                update_id=step["turn"],
            )
            results.append(self._evaluate_step(step, response))
        return {
            "case_id": case["case_id"],
            "category": case["category"],
            "steps": results,
            "passed": all(r["passed"] for r in results),
        }

    def _evaluate_step(self, step: dict, response: dict) -> dict:
        expected = step.get("expected", {})
        checks = {}

        if "intent" in expected:
            checks["intent_match"] = response["intent"] == expected["intent"]

        if "tools_called" in expected:
            actual_tools = set(response["tool_calls"])
            expected_tools = set(expected["tools_called"])
            checks["tools_called_match"] = expected_tools.issubset(actual_tools)

        if "state_after" in expected:
            checks["state_match"] = response["state"] == expected["state_after"]

        if "response_contains" in expected:
            text = response["response_text"]
            checks["response_contains"] = all(
                s.lower() in text.lower() for s in expected["response_contains"]
            )

        if "forbidden_tools" in expected:
            actual_tools = set(response["tool_calls"])
            forbidden = set(expected["forbidden_tools"])
            checks["no_forbidden_tool"] = not (actual_tools & forbidden)

        return {
            "turn": step["turn"],
            "checks": checks,
            "passed": all(checks.values()),
        }

    async def run_all(self) -> dict:
        cases = self.load_cases()
        results = await asyncio.gather(*[self.run_case(c) for c in cases])
        return {
            "total": len(results),
            "passed": sum(1 for r in results if r["passed"]),
            "failed": sum(1 for r in results if not r["passed"]),
            "cases": results,
        }
```

## 6. Metrics & Gates

### 6.1 Metrics

| Metric | Formula | Target Fase 1 | Target Fase 3 |
|:--|:--|:--|:--|
| Accuracy | passed_cases / total_cases | ≥ 90% | ≥ 95% |
| Business Safety | safe_cases / total_cases | 100% | 100% |
| No Forbidden Tool | cases_without_forbidden / total | 100% | 100% |
| Avg Turns | sum(turns) / total_cases | ≤ 5 | ≤ 4 |
| Intent F1 | harmonic_mean(precision, recall) | ≥ 0.85 | ≥ 0.92 |

### 6.2 CI Gate

```yaml
# CI fail condition
- accuracy < 0.90  → FAIL
- business_safety < 1.0  → FAIL (hard gate)
- no_forbidden_tool < 1.0  → FAIL (hard gate)
- regression vs main > 2%  → FAIL
```

### 6.3 Regression detection

```python
# evaluation/harness/metrics.py
def detect_regression(current: dict, baseline: dict, threshold: float = 0.02) -> list:
    """Deteksi regression vs baseline (main branch)."""
    regressions = []
    for category in current["by_category"]:
        curr_acc = current["by_category"][category]["accuracy"]
        base_acc = baseline["by_category"][category]["accuracy"]
        if curr_acc < base_acc - threshold:
            regressions.append({
                "category": category,
                "current": curr_acc,
                "baseline": base_acc,
                "delta": curr_acc - base_acc,
            })
    return regressions
```

## 7. CI Integration

### 7.1 run_evals.sh

```bash
#!/usr/bin/env bash
# evaluation/run_evals.sh
set -euo pipefail

cd "$(dirname "$0")/.."

# Setup
pip install -r evaluation/requirements.txt -q

# Run
python -m evaluation.harness.runner \
  --cases-dir evaluation/cases \
  --output evaluation/results/latest.json \
  --baseline evaluation/results/main.json \
  --gate-accuracy 0.90 \
  --gate-business-safety 1.0 \
  --gate-no-forbidden-tool 1.0 \
  --regression-threshold 0.02

# Generate report
python -m evaluation.harness.report \
  --input evaluation/results/latest.json \
  --output evaluation/results/latest.md
```

### 7.2 GitHub Actions integration

Update `.github/workflows/contract.yml` (atau buat `.github/workflows/eval.yml`):

```yaml
- name: Run evaluation harness
  run: bash evaluation/run_evals.sh

- name: Upload eval results
  uses: actions/upload-artifact@v4
  if: always()
  with:
    name: eval-results
    path: evaluation/results/

- name: Comment on PR (if failed)
  if: failure() && github.event_name == 'pull_request'
  uses: actions/github-script@v7
  with:
    script: |
      github.rest.issues.createComment({
        ...context.repo,
        issue_number: context.payload.pull_request.number,
        body: '❌ **Evaluation harness gagal.** Lihat artifact `eval-results` untuk detail.'
      })
```
